-- SHAGS | Cursor Hub
-- LocalScript em StarterPlayer > StarterPlayerScripts

local Players = game:GetService("Players")
local UIS = game:GetService("UserInputService")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- IDs dos assets
local LOGO_ID = "rbxassetid://104457688401830"
local SWORD_ID = "rbxassetid://99371372225483"
local CROSS_ID = "rbxassetid://94056530967449"

-- Estado
local selectedCursor = nil
local menuOpen = true
local centerLocked = false

local originalMouseIconEnabled = UIS.MouseIconEnabled
local originalMouseBehavior = UIS.MouseBehavior

-- GUI
local gui = Instance.new("ScreenGui")
gui.Name = "SHAGS_Hub"
gui.ResetOnSpawn = false
gui.IgnoreGuiInset = true
gui.Parent = playerGui

-- Janela principal
local main = Instance.new("Frame")
main.Name = "Main"
main.Size = UDim2.fromOffset(330, 300)
main.Position = UDim2.new(0.5, -165, 0.5, -150)
main.BackgroundColor3 = Color3.fromRGB(18, 18, 18)
main.BorderSizePixel = 0
main.Parent = gui

local mainCorner = Instance.new("UICorner")
mainCorner.CornerRadius = UDim.new(0, 12)
mainCorner.Parent = main

local mainStroke = Instance.new("UIStroke")
mainStroke.Color = Color3.fromRGB(75, 75, 75)
mainStroke.Thickness = 1.5
mainStroke.Parent = main

-- Barra superior
local top = Instance.new("Frame")
top.Size = UDim2.new(1, 0, 0, 65)
top.BackgroundColor3 = Color3.fromRGB(10, 10, 10)
top.BorderSizePixel = 0
top.Parent = main

local topCorner = Instance.new("UICorner")
topCorner.CornerRadius = UDim.new(0, 12)
topCorner.Parent = top

local logo = Instance.new("ImageLabel")
logo.Size = UDim2.fromOffset(48, 48)
logo.Position = UDim2.fromOffset(12, 8)
logo.BackgroundTransparency = 1
logo.Image = LOGO_ID
logo.Parent = top

local title = Instance.new("TextLabel")
title.Size = UDim2.new(1, -75, 1, 0)
title.Position = UDim2.fromOffset(68, 0)
title.BackgroundTransparency = 1
title.Text = "SHAGS"
title.TextColor3 = Color3.fromRGB(255, 255, 255)
title.Font = Enum.Font.GothamBold
title.TextSize = 23
title.TextXAlignment = Enum.TextXAlignment.Left
title.Parent = top

-- Arrastar janela pela barra superior
local dragging = false
local dragStart
local startPosition

top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = main.Position
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

-- Texto de seleção
local subtitle = Instance.new("TextLabel")
subtitle.Size = UDim2.new(1, -30, 0, 28)
subtitle.Position = UDim2.fromOffset(15, 78)
subtitle.BackgroundTransparency = 1
subtitle.Text = "SELECIONE SEU CURSOR"
subtitle.TextColor3 = Color3.fromRGB(190, 190, 190)
subtitle.Font = Enum.Font.GothamBold

subtitle.TextXAlignment = Enum.TextXAlignment.Left
subtitle.Parent = main

-- Cursor visual fixo no centro
local cursor = Instance.new("ImageLabel")
cursor.Name = "CustomCursor"
cursor.Size = UDim2.fromOffset(34, 34)
cursor.AnchorPoint = Vector2.new(0.5, 0.5)
cursor.Position = UDim2.fromScale(0.5, 0.5)
cursor.BackgroundTransparency = 1
cursor.Visible = false
cursor.ZIndex = 10
cursor.Parent = gui

-- Criar botões
local function createCursorButton(text, image, y)
    local button = Instance.new("TextButton")
    button.Size = UDim2.new(1, -30, 0, 48)
    button.Position = UDim2.fromOffset(15, y)
    button.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
    button.Text = text
    button.TextColor3 = Color3.fromRGB(255, 255, 255)
    button.Font = Enum.Font.GothamSemibold
    button.TextSize = 14
    button.AutoButtonColor = true
    button.Parent = main

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 8)
    corner.Parent = button

    local preview = Instance.new("ImageLabel")
    preview.Size = UDim2.fromOffset(30, 30)
    preview.Position = UDim2.new(1, -40, 0.5, -15)
    preview.BackgroundTransparency = 1
    preview.Image = image
    preview.Parent = button

    button.MouseButton1Click:Connect(function()
        selectedCursor = image
        cursor.Image = image
        cursor.Visible = true
        UIS.MouseIconEnabled = false
    end)

    return button
end

createCursorButton("Diamond Sword", SWORD_ID, 115)
createCursorButton("White Cross", CROSS_ID, 173)

-- Restaurar cursor original
local restore = Instance.new("TextButton")
restore.Size = UDim2.new(1, -30, 0, 40)
restore.Position = UDim2.fromOffset(15, 235)
restore.BackgroundColor3 = Color3.fromRGB(55, 55, 55)
restore.Text = "Restaurar cursor original"
restore.TextColor3 = Color3.fromRGB(255, 255, 255)
restore.Font = Enum.Font.GothamSemibold
restore.TextSize = 13
restore.Parent = main

local restoreCorner = Instance.new("UICorner")
restoreCorner.CornerRadius = UDim.new(0, 8)
restoreCorner.Parent = restore

restore.MouseButton1Click:Connect(function()
    selectedCursor = nil
    cursor.Visible = false
    UIS.MouseIconEnabled = originalMouseIconEnabled
    UIS.MouseBehavior = originalMouseBehavior
    centerLocked = false
end)

-- Abrir/fechar menu com RightShift
local function setMenuOpen(open)
    menuOpen = open
    main.Visible = open

    if open then
        UIS.MouseBehavior = Enum.MouseBehavior.Default
        centerLocked = false
    elseif selectedCursor then
        UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
        centerLocked = true
    end
end

UIS.InputBegan:Connect(function(input, processed)
    if processed then return end

    if input.KeyCode == Enum.KeyCode.RightShift then
        setMenuOpen(not menuOpen)
    end
end)

-- Manter o cursor visual no centro
RunService.RenderStepped:Connect(function()
    if cursor.Visible then
        cursor.Position = UDim2.fromScale(0.5, 0.5)
    end
end)

-- Libera o mouse ao remover o script
script.Destroying:Connect(function()
    UIS.MouseIconEnabled = originalMouseIconEnabled
    UIS.MouseBehavior = originalMouseBehavior
    gui:Destroy()
end)

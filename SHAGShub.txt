-- ARRASTAR MENU
local dragging = false
local dragStart
local startPosition

top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = main.Position
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then

        local delta = input.Position - dragStart

        main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = false
    end
end)

-- CURSOR VISUAL

cursor.Size = UDim2.fromOffset(cursorSize, cursorSize)
cursor.AnchorPoint = Vector2.new(0.5, 0.5)

cursor.BackgroundTransparency = 1
cursor.Visible = false
cursor.ZIndex = 100
cursor.Parent = gui

-- ATUALIZAR MODO DO CURSOR
local function updateCursorMode()
    if selectedCursor then
        cursor.Visible = not menuOpen
        UIS.MouseIconEnabled = menuOpen and true or false

        if menuOpen then
            UIS.MouseBehavior = Enum.MouseBehavior.Default
        else
            UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
        end
    else
        cursor.Visible = false
        UIS.MouseIconEnabled = originalMouseIconEnabled
        UIS.MouseBehavior = originalMouseBehavior
    end
end

-- BOTÕES DE CURSOR
local function createCursorButton(text, image, y)
    local button = Instance.new("TextButton")
    button.Size = UDim2.new(1, -30, 0, 48)
    button.Position = UDim2.fromOffset(15, y)
    button.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
    button.Text = text
    button.TextColor3 = Color3.fromRGB(255, 255, 255)
    button.Font = Enum.Font.GothamSemibold
    button.TextSize = 14
    button.Parent = main

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 8)
    corner.Parent = button

    local preview = Instance.new("ImageLabel")
    preview.Size = UDim2.fromOffset(30, 30)
    preview.Position = UDim2.new(1, -40, 0.5, -15)
    preview.BackgroundTransparency = 1
    preview.Image = image
    preview.Parent = button

    button.MouseButton1Click:Connect(function()
        selectedCursor = image
        cursor.Image = image
        cursor.Size = UDim2.fromOffset(cursorSize, cursorSize)
        updateCursorMode()
    end)

    return button
end

local decreaseButton = createSizeButton(
    "−",
    UDim2.new(0.5, -75, 0, 258)
)

local increaseButton = createSizeButton(
    "+",
    UDim2.new(0.5, 33, 0, 258)
)

local function updateSize()
    cursor.Size = UDim2.fromOffset(cursorSize, cursorSize)
    sizeDisplay.Text = tostring(cursorSize)
end

decreaseButton.MouseButton1Click:Connect(function()
    cursorSize = math.max(MIN_SIZE, cursorSize - SIZE_STEP)
    updateSize()
end)

increaseButton.MouseButton1Click:Connect(function()
    cursorSize = math.min(MAX_SIZE, cursorSize + SIZE_STEP)
    updateSize()
end)

-- ABRIR E FECHAR MENU
local function setMenuOpen(open)
    menuOpen = open
    main.Visible = open
    updateCursorMode()
end

closeButton.MouseButton1Click:Connect(function()
    setMenuOpen(false)
end)

UIS.InputBegan:Connect(function(input, processed)
    if input.KeyCode == Enum.KeyCode.RightShift then
        setMenuOpen(not menuOpen)
    end
end)

-- MANTER CURSOR NO CENTRO

    if cursor.Visible then
        cursor.Position = UDim2.fromScale(0.5, 0.5)
    end
end)
